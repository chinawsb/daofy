function TMainForm.ControlsToYoloLabel: TArray<string>;
var
  I: Integer;
  Ctrl: TControl;
  Cid: Integer;
  FW, FH: Single;
  XC, YC, NW, NH: Single;
  ResultList: TStringList;
begin
  FW := Self.ClientWidth;
  FH := Self.ClientHeight;
  if (FW <= 0) or (FH <= 0) then
    Result := nil;
    Exit;

  ResultList := TStringList.Create;
  try
    for I := 0 to Self.ControlCount - 1 do
    begin
      Ctrl := Self.Controls[I];
      if not Ctrl.Visible then
        Continue;

      Cid := GetClassId(Ctrl.ClassName);
      if Cid < 0 then
        Continue;

      // YOLO 格式: <class_id> <x_center> <y_center> <width> <height>
      XC := (Ctrl.Left + Ctrl.Width / 2) / FW;
      YC := (Ctrl.Top + Ctrl.Height / 2) / FH;
      NW := Ctrl.Width / FW;
      NH := Ctrl.Height / FH;

      // 边界裁剪
      XC := IfThen(XC < 0.0, 0.0, IfThen(XC > 1.0, 1.0, XC));
      YC := IfThen(YC < 0.0, 0.0, IfThen(YC > 1.0, 1.0, YC));
      NW := IfThen(NW < 0.0, 0.0, IfThen(NW > 1.0, 1.0, NW));
      NH := IfThen(NH < 0.0, 0.0, IfThen(NH > 1.0, 1.0, NH));

      ResultList.Add(Format('%d %.6f %.6f %.6f %.6f', [Cid, XC, YC, NW, NH]));
    end;
    Result := ResultList.ToStringArray;
  finally
    ResultList.Free;
  end;
end;
